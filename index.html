// ⚠️ MASUKKAN URL WEB APP GOOGLE APPS SCRIPT ANDA DI SINI const GAS_URL = "URL_WEB_APP_GOOGLE_APPS_SCRIPT_ANDA_DI_SINI"; // Variabel Global var session = null; var isMasuk = false; // Set tanggal hari ini otomatis di input form document.getElementById('tanggal').valueAsDate = new Date(); // ========================================== // 1. FUNGSI LOGIN VIA FETCH (API HTTP) // ========================================== function login() { const user = document.getElementById('user').value; const pass = document.getElementById('pass').value; const msg = document.getElementById('login-msg'); if(!user || !pass) { msg.innerText = "Username dan password wajib diisi!"; return; } msg.innerText = "Memverifikasi enkripsi keamanan..."; fetch(GAS_URL, { method: "POST", body: JSON.stringify({ action: "checkLogin", username: user, password: pass }) }) .then(res => res.json()) .then(res => { if(res.success) { session = res; // Simpan data login ke session global document.getElementById('login-page').classList.add('d-none'); document.getElementById('main-page').classList.remove('d-none'); document.getElementById('display-role').innerText = session.role; // PROTEKSI BARANG MASUK: HANYA MUNCUL JIKA ROLE BUKAN STAF var menuMasuk = document.getElementById("menuBarangMasuk"); if (menuMasuk) { if (session.role === "Staf") { menuMasuk.classList.add('d-none'); } else { menuMasuk.classList.remove('d-none'); } } showPage('dashboard'); } else { msg.innerText = "Kombinasi Username & Password salah!"; } }) .catch(err => { msg.innerText = "Gagal terhubung ke server! Periksa koneksi atau URL GAS."; console.error(err); }); } // ========================================== // 2. FUNGSI NAVIGASI HALAMAN // ========================================== function showPage(page) { document.getElementById('page-dashboard').classList.add('d-none'); document.getElementById('page-form').classList.add('d-none'); if(page === 'dashboard') { document.getElementById('page-dashboard').classList.remove('d-none'); } else if(page === 'masuk') { isMasuk = true; document.getElementById('page-form').classList.remove('d-none'); document.getElementById('form-title').innerText = "Form Barang Masuk"; let btn = document.getElementById('btn-submit'); btn.className = "btn btn-success w-100 fw-bold py-2 shadow-sm pointer"; } else if(page === 'keluar') { isMasuk = false; document.getElementById('page-form').classList.remove('d-none'); document.getElementById('form-title').innerText = "Form Barang Keluar"; let btn = document.getElementById('btn-submit'); btn.className = "btn btn-danger w-100 fw-bold py-2 shadow-sm pointer"; } } // ========================================== // 3. FUNGSI PROSES TRANSAKSI // ========================================== function prosesTransaksi() { const tipeTransaksi = isMasuk ? "MASUK" : "KELUAR"; const tanggal = document.getElementById('tanggal').value; const kode = document.getElementById('kode_barang').value; const jumlah = document.getElementById('jumlah').value; const ket = document.getElementById('keterangan').value; const msg = document.getElementById('trans-msg'); if (!tanggal || !kode || !jumlah) { alert("Mohon lengkapi Tanggal, Kode SKU, dan Jumlah Barang!"); return; } msg.innerText = "Sedang memproses ke sistem..."; const dataTransaksi = { tanggal: tanggal, kode: kode, jumlah: Number(jumlah), ket: ket, vendor: "" }; // Ambil nama user otomatis dari data session login const userYangLogin = session ? (session.username || session.user) : "Unknown"; fetch(GAS_URL, { method: "POST", body: JSON.stringify({ action: "submitTransaction", type: tipeTransaksi, data: dataTransaksi, user: userYangLogin }) }) .then(res => res.text()) .then(responseText => { alert(responseText); msg.innerText = ""; // Reset Form document.getElementById('kode_barang').value = ""; document.getElementById('jumlah').value = ""; document.getElementById('keterangan').value = ""; }) .catch(err => { alert("Terjadi kesalahan pengiriman!"); msg.innerText = ""; console.error(err); }); } // ========================================== // 4. FUNGSI LOGOUT // ========================================== function logout() { session = null; document.getElementById('user').value = ""; document.getElementById('pass').value = ""; document.getElementById('login-msg').innerText = ""; document.getElementById('main-page').classList.add('d-none'); document.getElementById('login-page').classList.remove('d-none'); }
